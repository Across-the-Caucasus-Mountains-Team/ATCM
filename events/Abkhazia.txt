add_namespace = abkhazia

#Shamil basayev arrived to abkhazia
country_event = {
	id = abkhazia.1
	title = abkhazia.1.t
	desc = abkhazia.1.d
	picture = GFX_Report_event_CHE_shamil_basayev

	is_triggered_only = yes

	option = {
		name = abkhazia.1.a
		add_war_support = 0.025
		if = {
			limit = { NOT = { has_character = ABK_shamil_basayev } }
			recruit_character = ABK_shamil_basayev
		}
		add_manpower = 75
		add_equipment_to_stockpile = { 
			type = infantry_equipment_1 
			amount = 75 
		}
	}
}
