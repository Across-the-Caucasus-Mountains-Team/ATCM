on_actions = {
	on_monthly = {
		effect = {
			ATCM_economy_setup = yes
			add_to_variable = { cash = var:income }
			subtract_from_variable = { cash = var:expense }
			multiply_variable = { cash = 10 }
			round_variable = cash
			divide_variable = { cash = 10 }
		}
	}

	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = { tag = CHE FROM = { tag = SAK } }
						AND = { tag = SAK FROM = { tag = CHE } }
					}
					NOT = { has_global_flag = CHE_georgia_war_news_active }
				}
				set_global_flag = CHE_georgia_war_news_active
				news_event = { id = news.34 hours = 1 }
			}
			if = {
				limit = {
					OR = {
						AND = { tag = CHE FROM = { tag = SAK } }
						AND = { tag = SAK FROM = { tag = CHE } }
					}
					CHE = {
						has_completed_focus = CHE_light_a_fire_in_georgia
						NOT = { has_country_flag = CHE_georgia_operation_resolved }
						NOT = { has_country_flag = CHE_georgia_operation_pending }
					}
				}
				CHE = {
					if = {
						limit = { NOT = { has_country_flag = CHE_georgia_resistance_snapshot } }
						set_variable = { CHE_georgia_resistance_at_war = 0 }
						if = {
							limit = { 34 = { CHE_georgia_unrest_state = yes } }
							add_to_variable = { CHE_georgia_resistance_at_war = 34.resistance }
						}
						if = {
							limit = { 120 = { CHE_georgia_unrest_state = yes } }
							add_to_variable = { CHE_georgia_resistance_at_war = 120.resistance }
						}
						divide_variable = { CHE_georgia_resistance_at_war = 2 }
						set_country_flag = CHE_georgia_resistance_snapshot
					}
					set_country_flag = CHE_georgia_operation_pending
					country_event = { id = chechnya.83 hours = 1 }
				}
			}
		}
	}
	on_weekly = {
		effect = {
			if = {
				limit = {
					tag = CHE
					NOT = { has_war_with = SAK }
					OR = {
						has_country_flag = CHE_GelayevInGeorgia
						has_country_flag = CHE_georgia_operation_pending
						has_country_flag = CHE_georgia_resistance_snapshot
					}
				}
				if = {
					limit = {
						OR = {
							has_country_flag = CHE_georgia_operation_resolved
							has_country_flag = CHE_georgia_operation_pending
							NOT = { country_exists = SAK }
						}
					}
					clr_country_flag = CHE_GelayevInGeorgia
					clr_country_flag = CHE_georgia_operation_pending
					clr_country_flag = CHE_georgia_resistance_snapshot
				}
			}
		}
	}

	on_daily = {
		effect = {
			if = {
				limit = {
					has_global_flag = CHE_georgia_war_news_active
					NOT = { CHE = { has_war_with = SAK } }
				}
				clr_global_flag = CHE_georgia_war_news_active
				news_event = { id = news.35 hours = 1 }
			}
			if = {
				limit = {
					OR = { tag = CHE tag = SAK }
					CHE = { has_war_with = SAK }
					NOT = { has_global_flag = CHE_georgia_war_news_active }
				}
				set_global_flag = CHE_georgia_war_news_active
				news_event = { id = news.34 hours = 1 }
			}
			if = {
				limit = { tag = CHE has_country_flag = CHE_georgia_peace_offer_pending }
				if = {
					limit = {
						OR = {
							NOT = { has_war_with = SAK }
							NOT = { country_exists = SAK }
							has_country_flag = { flag = CHE_georgia_peace_offer_pending days > 29 }
						}
					}
					clr_country_flag = CHE_georgia_peace_offer_pending
					clr_country_flag = CHE_georgia_peace_offer_dispatched
					country_event = { id = chechnya.86 hours = 1 }
				}
				else_if = {
					limit = { NOT = { has_country_flag = CHE_georgia_peace_offer_dispatched } }
					clr_country_flag = CHE_georgia_peace_offer_pending
					set_country_flag = CHE_georgia_peace_offer_pending
					set_country_flag = CHE_georgia_peace_offer_dispatched
					SAK = { country_event = { id = Georgia.58 hours = 1 } }
				}
			}
			if = {
				limit = {
					tag = CHE
					has_country_flag = CHE_azerbaijan_operation_started
					OR = { NOT = { has_country_flag = CHE_azerbaijan_campaign_finished } has_country_flag = CHE_azerbaijan_peace_offer_pending }
				}
				CHE_update_azerbaijan_campaign_effect = yes
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					FROM = { tag = CHE }
					FROM.FROM = { has_state_flag = CHE_azerbaijan_prewar_state }
					CHE = { has_war_with = AZR has_country_flag = CHE_azerbaijan_operation_started NOT = { has_country_flag = CHE_azerbaijan_campaign_finished } }
				}
				CHE = { CHE_check_azerbaijan_retreat_effect = yes }
			}
		}
	}
}
