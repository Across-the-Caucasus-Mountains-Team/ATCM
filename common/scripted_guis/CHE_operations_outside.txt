scripted_gui = {

    CHE_operations_decision_window = {
       context_type = decision_category
	   window_name = "CHE_operations_decision_window"
 
        visible = {
            tag = CHE
        }
 
		effects = {
			operations_outside_button_click = {
				if = {
					limit = { has_variable = CHE_show_operations_outside }
					clear_variable = CHE_show_operations_outside
				}
				else = {
					set_variable = { CHE_show_operations_outside = 1 }
				}
			}
		}
        triggers = {
        }
		
		properties = {
			
		}
    }

    CHE_operations_window = {
       context_type = player_context 
	   window_name = "CHE_operations_window"
 
        visible = {
            tag = CHE
            check_variable = { CHE_show_operations_outside > 0 }
        }
 
		effects = {
			close_operations_outside_button_click = {
				set_variable = { CHE_show_operations_outside = 0 }
			}
		}

        triggers = {

        }
		
		properties = {
			
		}
    }
}
