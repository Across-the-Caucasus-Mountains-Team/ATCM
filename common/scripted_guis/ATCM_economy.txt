		window_name = "economy_button_window"
		context_type = player_context
		parent_window_token = top_bar

		triggers = {
			cash_button_visible = {
				check_variable = {
					var = cash
					value = 0
					compare = greater_than_or_equals
				}
			}
			debt_button_visible = {
				check_variable = {
					cash < 0
				}
			}
		}

		effects = {
			cash_button_click = {
				ATCM_economy_setup = yes
				if = {
					limit = {
						has_country_flag = economy_tab_on
					}
					clr_country_flag = economy_tab_on
					clr_country_flag = economy_decisions_tab_on
				}
				else = {
					set_country_flag = economy_tab_on
				}
			}
