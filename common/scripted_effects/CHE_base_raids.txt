CHE_initialize_base_raids_effect = {
	if = {
		limit = { NOT = { has_variable = CHE_base_taken_military } }
		set_variable = { CHE_base_taken_military = 0 }
	}
	if = {
		limit = { NOT = { has_variable = CHE_base_taken_vehicles } }
		set_variable = { CHE_base_taken_vehicles = 0 }
	}
	if = {
		limit = { NOT = { has_variable = CHE_base_taken_airfields } }
		set_variable = { CHE_base_taken_airfields = 0 }
	}
	if = {
		limit = { NOT = { has_variable = CHE_base_taken_heavy } }
		set_variable = { CHE_base_taken_heavy = 0 }
	}
}

CHE_base_item_amount_effect = {
	set_temp_variable = { base_raid_item_amount = var:base_raid_new_share }
	multiply_temp_variable = { base_raid_item_amount = var:base_raid_item_total }
	round_temp_variable = base_raid_item_amount
	set_temp_variable = { base_raid_previous_amount = var:base_raid_old_share }
	multiply_temp_variable = { base_raid_previous_amount = var:base_raid_item_total }
	round_temp_variable = base_raid_previous_amount
	subtract_from_temp_variable = { base_raid_item_amount = var:base_raid_previous_amount }
	if = {
		limit = { has_country_flag = CHE_base_record_handover }
		add_to_array = { array = base_raid_handover value = var:base_raid_item_amount }
	}
}

CHE_base_raid_share_effect = {
	if = {
		limit = { check_variable = { var = base_raid_old_share value = 0 compare = equals } }
		set_temp_variable = { base_raid_share = 0.4 }
	}
	else = { set_temp_variable = { base_raid_share = 0.2 } }
	set_temp_variable = { base_raid_new_share = var:base_raid_old_share }
	add_to_temp_variable = { base_raid_new_share = var:base_raid_share }
	clamp_temp_variable = { var = base_raid_new_share min = 0 max = 1 }
}

CHE_cancel_base_raid_effect = {
	if = {
		limit = { has_country_flag = CHE_base_raid_active }
		clr_country_flag = CHE_base_raid_active
		add_political_power = 25
	}
}

CHE_base_raid_success_effect = {
	if = {
		limit = { check_variable = { var = base_raid_target value = 0 compare = equals } }
		CHE_raid_base_military_effect = yes
	}
	if = {
		limit = { check_variable = { var = base_raid_target value = 1 compare = equals } }
		CHE_raid_base_vehicles_effect = yes
	}
	if = {
		limit = { check_variable = { var = base_raid_target value = 2 compare = equals } }
		CHE_raid_base_airfields_effect = yes
	}
	if = {
		limit = { check_variable = { var = base_raid_target value = 3 compare = equals } }
		CHE_raid_base_heavy_effect = yes
	}
}

CHE_finish_base_raid_effect = {
	if = {
		limit = {
			has_country_flag = CHE_base_raid_active
			CHE_base_raid_can_run = yes
		}
		clr_country_flag = CHE_base_raid_active
		random_list = {
			90 = { CHE_base_raid_success_effect = yes }
			10 = {
				add_manpower = -25
				country_event = atcm_base_raids.2
			}
		}
	}
	else = { CHE_cancel_base_raid_effect = yes }
}

CHE_base_handover_effect = {
	if = {
		limit = {
			tag = CHE
			NOT = { has_country_flag = CHE_base_handover_completed }
		}
		CHE_initialize_base_raids_effect = yes
		set_country_flag = CHE_base_handover_completed
		CHE_cancel_base_raid_effect = yes
		remove_decision = CHE_raid_base_military
		remove_decision = CHE_raid_base_vehicles
		remove_decision = CHE_raid_base_airfields
		remove_decision = CHE_raid_base_heavy
		clear_array = base_raid_handover
		set_country_flag = CHE_base_record_handover
		set_temp_variable = { base_raid_old_share = var:CHE_base_taken_military }
		set_temp_variable = { base_raid_new_share = 1 }
		add_to_temp_variable = { base_raid_new_share = var:CHE_base_taken_military }
		divide_temp_variable = { base_raid_new_share = 2 }
		CHE_award_base_military_effect = yes
		set_temp_variable = { base_raid_old_share = var:CHE_base_taken_vehicles }
		set_temp_variable = { base_raid_new_share = 1 }
		add_to_temp_variable = { base_raid_new_share = var:CHE_base_taken_vehicles }
		divide_temp_variable = { base_raid_new_share = 2 }
		CHE_award_base_vehicles_effect = yes
		set_temp_variable = { base_raid_old_share = var:CHE_base_taken_airfields }
		set_temp_variable = { base_raid_new_share = 1 }
		add_to_temp_variable = { base_raid_new_share = var:CHE_base_taken_airfields }
		divide_temp_variable = { base_raid_new_share = 2 }
		CHE_award_base_airfields_effect = yes
		set_temp_variable = { base_raid_old_share = var:CHE_base_taken_heavy }
		set_temp_variable = { base_raid_new_share = 1 }
		add_to_temp_variable = { base_raid_new_share = var:CHE_base_taken_heavy }
		divide_temp_variable = { base_raid_new_share = 2 }
		CHE_award_base_heavy_effect = yes
		clr_country_flag = CHE_base_record_handover
		if = {
			limit = { country_exists = RAS }
			RAS = {
				clear_array = base_raid_handover
				set_country_flag = CHE_base_record_handover
				set_temp_variable = { base_raid_old_share = 1 }
				add_to_temp_variable = { base_raid_old_share = CHE.CHE_base_taken_military }
				divide_temp_variable = { base_raid_old_share = 2 }
				set_temp_variable = { base_raid_new_share = 1 }
				CHE_award_base_military_effect = yes
				set_temp_variable = { base_raid_old_share = 1 }
				add_to_temp_variable = { base_raid_old_share = CHE.CHE_base_taken_vehicles }
				divide_temp_variable = { base_raid_old_share = 2 }
				set_temp_variable = { base_raid_new_share = 1 }
				CHE_award_base_vehicles_effect = yes
				set_temp_variable = { base_raid_old_share = 1 }
				add_to_temp_variable = { base_raid_old_share = CHE.CHE_base_taken_airfields }
				divide_temp_variable = { base_raid_old_share = 2 }
				set_temp_variable = { base_raid_new_share = 1 }
				CHE_award_base_airfields_effect = yes
				set_temp_variable = { base_raid_old_share = 1 }
				add_to_temp_variable = { base_raid_old_share = CHE.CHE_base_taken_heavy }
				divide_temp_variable = { base_raid_old_share = 2 }
				set_temp_variable = { base_raid_new_share = 1 }
				CHE_award_base_heavy_effect = yes
				clr_country_flag = CHE_base_record_handover
				set_country_flag = CHE_base_handover_received
				country_event = { id = atcm_base_raids.1 hours = 1 }
			}
		}
	}
