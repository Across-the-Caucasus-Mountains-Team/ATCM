CHE_vol_init_effect = {
	if = {
		limit = { NOT = { has_variable = CHE_fv_org } }
		set_variable = { CHE_fv_org = 0.02 }
	}
	if = {
		limit = { NOT = { has_variable = CHE_fv_mp } }
		set_variable = { CHE_fv_mp = 5 }
	}
	if = {
		limit = { NOT = { has_variable = CHE_fv_pp } }
		set_variable = { CHE_fv_pp = -0.02 }
	}
	if = {
		limit = { NOT = { has_variable = CHE_fv_plan } }
		set_variable = { CHE_fv_plan = 0.05 }
	}
	if = {
		limit = { NOT = { has_variable = CHE_unso_men } }
		set_variable = { CHE_unso_men = 0 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = CHE_foreign_volunteers_dm } } }
		add_dynamic_modifier = { modifier = CHE_foreign_volunteers_dm }
	}
}

CHE_vol_boost_effect = {
	CHE_vol_init_effect = yes
	add_to_variable = { CHE_fv_org = 0.02 }
	add_to_variable = { CHE_fv_mp = 5 }
	add_to_variable = { CHE_fv_pp = -0.02 }
	add_to_variable = { CHE_fv_plan = 0.05 }
}
