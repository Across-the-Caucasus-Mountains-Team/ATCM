ATCM_increase_education_spending = {
	if = {
		limit = { has_idea = private_education }
		add_ideas = free_elementary_and_middle_schools
	}
	else_if = {
		limit = { has_idea = free_elementary_and_middle_schools }
		add_ideas = free_high_schools
	}
	else_if = {
		limit = { has_idea = free_high_schools }
		add_ideas = free_universities
	}
	else_if = {
		limit = { has_idea = free_universities }
		add_ideas = GEN_university_labs
	}
	else_if = {
		limit = { has_idea = GEN_university_labs }
		add_research_slot = 1
	}
	ATCM_economy_setup = yes
}

ATCM_decrease_education_spending = {
	if = {
		limit = { has_idea = private_education }
		add_research_slot = -1
	}
	else_if = {
		limit = { has_idea = free_elementary_and_middle_schools }
		add_ideas = private_education
	}
	else_if = {
		limit = { has_idea = free_high_schools }
		add_ideas = free_elementary_and_middle_schools
	}
	else_if = {
		limit = { has_idea = free_universities }
		add_ideas = free_high_schools
	}
	else_if = {
		limit = { has_idea = GEN_university_labs }
		remove_ideas = GEN_university_labs
	}
	ATCM_economy_setup = yes
}

upgrade_economy_law = {
	if = {
		limit = { has_idea = civilian_economy }
		add_ideas = low_economic_mobilisation
	}
	else_if = {
		limit = { has_idea = low_economic_mobilisation }
		add_ideas = partial_economic_mobilisation
	}
	else_if = {
		limit = { has_idea = partial_economic_mobilisation }
		add_ideas = war_economy
	}
	else_if = {
		limit = { has_idea = war_economy }
		add_ideas = tot_economic_mobilisation
	}
	ATCM_economy_setup = yes
}
