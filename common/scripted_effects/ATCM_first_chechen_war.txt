	RAS = {
		clr_country_flag = VDVLandingInProgress
		remove_mission = RAS_failed_blitzkrieg_decision
		RAS_finish_airborne_operation_effect = yes
		remove_mission = RAS_take_Grozny
		remove_mission = RAS_take_Shali
		remove_mission = RAS_take_Gudermes
		remove_mission = RAS_take_UrusMartan
		remove_mission = RAS_take_Argun
		mark_focus_tree_layout_dirty = yes
	}
	CHE = {
		clr_country_flag = FirstChechenWar
		remove_mission = CHE_hold_Grozny
		remove_mission = CHE_hold_Shali
		remove_mission = CHE_hold_Gudermes
		remove_mission = CHE_hold_UrusMartan
		remove_mission = CHE_hold_Argun
		remove_mission = CHE_monthly_weapon_supplies
		remove_mission = CHE_return_Grozny
		remove_ideas = {
			CHE_first_chechen_war_idea
			CHE_first_chechen_war_idea_2
		}
		mark_focus_tree_layout_dirty = yes

	DAG = {

		}
	}

RAS_start_airborne_operation_effect = {
	if = {
		limit = {
			has_war_with = CHE
			has_completed_focus = RAS_vdv_operation
			NOT = { has_country_flag = RAS_airborne_operation_started }
		}
		set_country_flag = RAS_airborne_operation_started
		set_country_flag = RAS_samashki_vdv_unlocked
		if = {
			limit = { has_country_flag = VDVLanding_at_Grozny_Airport_Prepared }
			set_country_flag = RAS_airport_landing_pending
			country_event = { id = russia.46 hours = 2 }
		}
		if = {
			limit = { has_country_flag = RAS_terek_crossings_prepared }
			country_event = { id = russia.48 hours = 3 }
		}
		if = {
			limit = { has_country_flag = RAS_thunder_prepared }
			country_event = { id = russia.49 hours = 24 }
		}
		clr_country_flag = VDVLandingInProgress
	}
}

RAS_finish_airborne_operation_effect = {
	if = {
		limit = { has_country_flag = RAS_airborne_operation_started }
		set_country_flag = RAS_airborne_operation_finished
		clr_country_flag = VDVLandingInProgress
		clr_country_flag = RAS_airport_landing_pending
		clr_country_flag = RAS_airport_hold_pending
		clr_country_flag = RAS_airport_reinforcements_unlocked
		clr_country_flag = RAS_samashki_vdv_unlocked
		clr_country_flag = RAS_khankala_flight_pending
		clr_country_flag = RAS_terek_crossings_prepared
		clr_country_flag = RAS_thunder_prepared
	}
}

RAS_fail_grozny_airport_operation_effect = {
	if = {
		limit = {
			has_country_flag = RAS_airport_hold_pending
			NOT = { has_country_flag = RAS_airport_landing_failed }
		}
		clr_country_flag = RAS_airport_hold_pending
		clr_country_flag = RAS_airport_reinforcements_unlocked
		set_country_flag = RAS_airport_landing_failed
		country_event = { id = russia.52 }
	}
}
